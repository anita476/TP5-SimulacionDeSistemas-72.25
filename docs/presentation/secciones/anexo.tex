\begin{frame}{Geometría de la capa de elusión}
  \large
  Marco local de $i$, con $\hat{\vv{u}}$ hacia el target final y $\hat{\vv{n}} = (u_y,\,-u_x)$:
  \[
    s = (\vv{r}_j - \vv{r}_i)\cdot\hat{\vv{u}}, \qquad \ell = (\vv{r}_j - \vv{r}_i)\cdot\hat{\vv{n}}
  \]
  \vspace{2pt}
  Parábola por $i$, punto de paso $P = (s,\ \ell + R)$ y target final:
  \[
    y(x) = (\ell + R)\,\frac{x\,(D - x)}{s\,(D - s)}
    \qquad\Rightarrow\qquad
    y'(0) = \frac{(\ell + R)\,D}{s\,(D - s)}
  \]
  \vspace{2pt}
  Target temporal, sobre tangente a parábola en $i$:
  \[
    \vv{x}^{\,temp}_i = \vv{r}_i + s\,\hat{\vv{u}} + s\,y'(0)\,\hat{\vv{n}},
    \qquad
    \vv{e}_0 = \frac{\vv{x}^{\,temp}_i - \vv{r}_i}{|\vv{x}^{\,temp}_i - \vv{r}_i|}
  \]
\end{frame}

\begin{frame}{Distribución de $|\delta|$}
  \begin{cols}
    \begin{column}{0.62\textwidth}
      \figura{desvios_distribucion.png}
    \end{column}
    \begin{column}{0.34\textwidth}
      \begin{parametros}
        \item $18$ eventos, $36$ peatones
        \item $|\delta| = \max_k \big|(\vv{r}_k - \vv{r}_0)\cdot\hat{\vv{n}}\big|$
        \item $\hat{\vv{n}}$: normal a $\vv{r}_f - \vv{r}_0$, hacia la derecha del peatón
        \item Ancho de bin $0.05\uni{m}$
      \end{parametros}
      \vspace{6pt}
      \begin{resultado}
        \item Ningún peatón entre $0.15$ y $0.20\uni{m}$: umbral de elusión $0.15\uni{m}$
        \item $6$ peatones por debajo del umbral
      \end{resultado}
    \end{column}
  \end{cols}
\end{frame}
