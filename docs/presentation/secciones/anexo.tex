
\begin{frame}{Distribución de $|\delta|$}
  \begin{cols}
    \begin{column}{0.62\textwidth}
      \figura{desvios_distribucion.png}
    \end{column}
    \begin{column}{0.34\textwidth}
      \begin{parametros}
        \item $18$ eventos, $36$ peatones
        \item $|\delta| = \max_k \big|(\vv{r}_i(t_k) - \vv{r}_i(t_0))\cdot\hat{\vv{n}}\big|$
        \item $\hat{\vv{n}}$: normal a la recta de primera a última posición, hacia derecha de peatón
        \item Ancho de bin $0.05\uni{m}$
      \end{parametros}
      \vspace{6pt}
      \begin{resultado}
        \item Umbral de elusión $0.15\uni{m}$
        \item $6$ peatones por debajo de umbral
      \end{resultado}
    \end{column}
  \end{cols}
\end{frame}
